## 3. Code thực tế

Đây là ví dụ cách lấy dữ liệu âm thanh:

javascript

```javascript

const audio = new Audio("music.mp3");

const audioContext = new AudioContext();

const source = audioContext.createMediaElementSource(audio);

const analyser = audioContext.createAnalyser();

analyser.fftSize = 2048;

source.connect(analyser);
analyser.connect(audioContext.destination);

const data = new Uint8Array(analyser.frequencyBinCount);

function analyzeMusic() {
    analyser.getByteFrequencyData(data);

    // Lấy năng lượng âm trầm
    let bass = 0;

    for (let i = 0; i < 10; i++) {
        bass += data[i];
    }

    bass /= 10 * 255;

    console.log("Bass:", bass);

    requestAnimationFrame(analyzeMusic);
}

// Gọi sau thao tác người dùng để mở AudioContext
audioContext.resume().then(() => {
    audio.play();
    analyzeMusic();
});
```

Giá trị `bass` thường nằm trong khoảng 0 đến 1.

Ví dụ:

- 0.1: âm trầm yếu.
- 0.5: âm trầm trung bình.
- 0.9: âm trầm mạnh.

Game sẽ dùng giá trị này để điều khiển hiệu ứng 3D.

```javascript
const geometry = new THREE.SphereGeometry(0.5, 32, 32);
```

```javascript
const effects = [
  { color: 0x00ffff, bloom: 1.5 },
  { color: 0xff00ff, bloom: 2.0 },
  { color: 0x00ff66, bloom: 1.8 },
  { color: 0xff6600, bloom: 2.5 }
];

let current = 0;

function changeEffect() {
  current = (current + 1) % effects.length;

  const effect = effects[current];

  // Đổi màu đường neon
  roadMaterial.color.setHex(effect.color);

  // Chuyển cường độ phát sáng
  // Có thể dùng GSAP để chuyển mượt
  gsap.to(bloomPass, {
    strength: effect.bloom,
    duration: 2
  });
}
```
